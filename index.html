} const incoming = new URL(request.url); const target = incoming.searchParams.get("url"); // SHAKIL VERSION CHECK if (!target) { return new Response(JSON.stringify({ success: true, worker: "SHAKIL", version: "V2", exchange: "MEXC", message: "SHAKIL V2 ACTIVE" }), { status: 200, headers: { ...corsHeaders, "Content-Type": "application/json;
