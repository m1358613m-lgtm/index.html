export default { async fetch(request) { const url = new URL(request.url); const targetUrl = 'https://contract.mexc.com' + url.pathname + url.search; try { const response = await fetch(targetUrl, { headers: { 'User-Agent': 'Mozilla/5.0' } }); const newHeaders = new Headers(response.headers); newHeaders.set('Access-Control-Allow-Origin', '*'); return new Response(response.body, { status: response.status, headers: newHeaders }); } catch (e) { return new Response('Proxy Error', { status: 500 }); } } };
