export default { async fetch(request) { const cors = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, OPTIONS", "Access-Control-Allow-Headers": "*", "Content-Type": "application/json; charset=utf-8" }; const json = (data, status = 200) => new Response(JSON.stringify(data), { status, headers: cors }); if (request.method === "OPTIONS") { return new Response(null, { status: 204, headers: cors }); } if (request.method !== "GET") { return json({ success: false, error: "GET only" }, 405); } const BASE = "https://contract.mexc.com"; const url = new URL(request.url); async function mexc(path) { const response = await fetch(BASE + path, { method: "GET", headers: { "Accept": "application/json", "User-Agent": "Mozilla/5.0" } }); const body = await response.text(); if (!response.ok) { throw new Error("MEXC HTTP " + response.status + ": " + body.slice(0, 500)); } let data; try { data = JSON.parse(body); } catch { throw new Error("Invalid MEXC JSON response"); } if (data.success === false) { throw new Error(data.message || data.msg || "MEXC API error"); } return data; } try { // Health check if (url.pathname === "/" || url.pathname === "/health") { return json({ success: true, worker: "SHAKIL", exchange: "MEXC", version: "COMPATIBLE-2", routes: ["/contracts", "/tickers", "/market", "/kline"], message: "Worker is running" }); } // Backward-compatible URL proxy const target = url.searchParams.get("url"); if (target) { const parsed = new URL(target); if ( parsed.protocol !== "https:" || parsed.hostname !== "contract.mexc.com" || !parsed.pathname.startsWith("/api/v1/") ) { return json({ success: false, error: "Invalid proxy URL" }, 403); } const response = await fetch(parsed.toString(), { headers: { "Accept": "application/json" } }); return new Response(response.body, { status: response.status, headers: { ...cors, "Content-Type": response.headers.get("Content-Type") || "application/json" } }); } // Contract list if (url.pathname === "/contracts") { return json(await mexc("/api/v1/contract/detail")); } // Tickers if (url.pathname === "/tickers") { return json(await mexc("/api/v1/contract/ticker")); } // Combined market data if (url.pathname === "/market") { const [contracts, tickers] = await Promise.all([ mexc("/api/v1/contract/detail"), mexc("/api/v1/contract/ticker") ]); return json({ success: true, contracts, tickers }); } // Candlesticks if (url.pathname === "/kline") { const symbol = url.searchParams.get("symbol") || "BTC_USDT"; const interval = url.searchParams.get("interval") || "Min5"; if (!/^[A-Z0-9_]{3,30}$/.test(symbol)) { return json({ success: false, error: "Invalid symbol" }, 400); } const allowed = [ "Min1", "Min5", "Min15", "Min30", "Min60", "Hour4", "Hour8", "Day1", "Week1", "Month1" ]; if (!allowed.includes(interval)) { return json({ success: false, error: "Invalid interval" }, 400); } const params = new URLSearchParams(); for (const key of ["start", "end"]) { const value = url.searchParams.get(key); if (value && /^\d+$/.test(value)) { params.set(key, value); } } const query = params.toString(); const path = "/api/v1/contract/kline/" + encodeURIComponent(symbol) + "?interval=" + encodeURIComponent(interval) + (query ? "&" + query : ""); return json(await mexc(path)); } return json({ success: false, error: "Unknown endpoint", available: ["/", "/health", "/contracts", "/tickers", "/market", "/kline"] }, 404); } catch (error) { return json({ success: false, worker: "SHAKIL", error: String(error.message || error) }, 502); } } };
